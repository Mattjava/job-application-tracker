{# Dashboard (home page). All statistics are worked out here in the template, so the route only has to pass the raw rows. Variables this template expects from the route: applications - rows from show_all_applications() jobs - rows from show_all_jobs() companies - rows from show_all_companies() contacts - rows from show_all_contacts() Status groups (Waiting / Interviewing / Offer / Closed) come from macros.html. #} {% extends "base.html" %} {% import "macros.html" as m %} {% block title %}Dashboard{% endblock %} {% block content %} {% set apps = applications | default([]) %} {% set all_jobs = jobs | default([]) %} {% set all_companies = companies | default([]) %} {% set all_contacts = contacts | default([]) %} {# Lookups so we can show "Job title at Company" instead of IDs #} {% set company_lookup = {} %} {% for company in all_companies %}{% set _ = company_lookup.update({company.company_id: company.company_name}) %}{% endfor %} {% set job_lookup = {} %} {% for job in all_jobs %}{% set _ = job_lookup.update({job.job_id: job}) %}{% endfor %} {# Application counts by pipeline stage #} {% set total = apps | length %} {% set waiting = apps | selectattr("status", "in", m.STATUS_GROUPS["Waiting"]) | list %} {% set interviewing = apps | selectattr("status", "in", m.STATUS_GROUPS["Interviewing"]) | list %} {% set offers = apps | selectattr("status", "in", m.STATUS_GROUPS["Offer"]) | list %} {% set closed = apps | selectattr("status", "in", m.STATUS_GROUPS["Closed"]) | list %} {% set cover_letters = apps | selectattr("cover_letter_sent") | list | length %} {# Rates: "heard back" = anything that has moved past Applied/Pending #} {% set heard_back = total - (waiting | length) %} {% set response_rate = (heard_back / total * 100) | round | int if total else 0 %} {% set interview_rate = ((interviewing | length + offers | length) / total * 100) | round | int if total else 0 %} {# Jobs #} {% set active_jobs = all_jobs | selectattr("is_active") | list %} {% set applied_job_ids = apps | map(attribute="job_id") | list %} {% set not_applied = active_jobs | rejectattr("job_id", "in", applied_job_ids) | list %} {% macro job_label(job_id) -%} {%- set job = job_lookup.get(job_id) -%} {%- if job -%}{{ job.job_title }} at {{ company_lookup.get(job.company_id, "Company #" ~ job.company_id) }} {%- else -%}Job #{{ job_id }}{%- endif -%} {%- endmacro %}
